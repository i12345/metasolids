// /**
//  * Algorithm:
//  * 
//  * Regret subdivisions that yielded no change in surface geometry,
//  * or recommend subdividing where there are quick changes.
//  * 
//  * One implementation:
//  *  for each newly-subdivided subcell:
//  *   compute the mean of:
//  *    for each edge it is connected to:
//  *     find the best matching edge from previous surface for the parent cell
//  *     compute a difference metric for how this edge changed,
//  *     in a way that would not make a difference if the cell it's connecting to
//  *     was subdivded or not
//  *    end for
//  *   compute mean
//  *   This mean edge-difference is recommends further subdivision
//  * 
//  * Surface mesh doesn't have to be limited to strictly interior cells, and
//  * interior cells are not directly the recommendation for subdivision.
//  * 
//  * Even exterior cells can be recommended for subdivision
//  */

export class VolumeSamplingSubdivisionSolidsBoundaryProcessor<
        IndicesT extends IndicesTypedArray = IndicesTypedArray,
        VolumeLocationT extends VolumeLocation = VolumeLocation,
        VolumeSampleT extends VolumeSample = VolumeSample,
        VolumeSamplingContextT extends
            VolumeSamplingContext<VolumeLocationT> =
            VolumeSamplingContext<VolumeLocationT>,
        VolumeSampleProcessingContextT = any,
        VolumeT extends
            VolumeWithBoundingBox<VolumeLocationT, VolumeSampleT, VolumeSamplingContextT> =
            VolumeWithBoundingBox<VolumeLocationT, VolumeSampleT, VolumeSamplingContextT>,
        VolumeProcessingT extends
            VolumeProcessingWithSamplingSolidsBoundary<
                    IndicesT,
                    VolumeLocationT,
                    VolumeSampleT,
                    VolumeSamplingContextT,
                    VolumeT
                > =
            VolumeProcessingWithSamplingSolidsBoundary<
                    IndicesT,
                    VolumeLocationT,
                    VolumeSampleT,
                    VolumeSamplingContextT,
                    VolumeT
                >,
        VolumeProcessingContextT extends
            VolumeProcessingContextWithSamplingSolidsBoundary<
                    IndicesT,
                    VolumeLocationT,
                    VolumeSampleT,
                    VolumeSamplingContextT,
                    VolumeSampleProcessingContextT
                > =
            VolumeProcessingContextWithSamplingSolidsBoundary<
                    IndicesT,
                    VolumeLocationT,
                    VolumeSampleT,
                    VolumeSamplingContextT,
                    VolumeSampleProcessingContextT
                >,
        > implements
    VolumeSamplingSubdivisionProcessor<
        IndicesT,
        BoundaryVolumeSamplingSubdivisionOctTreeGroups,
        BoundaryVolumeSamplingSubdivisionOctTreeValue,
        BoundaryVolumeSamplingSubdivisionOctTreeValuesGrouped,
        BoundaryVolumeSamplingSubdivisionOctTreeLayer, // <IndicesT>,
        BoundaryVolumeSamplingSubdivisionOctTreeLayersGrouped, // <IndicesT>,
        BoundaryVolumeSamplingSubdivisionOctTreesGrouped, // <IndicesT>,
        VolumeLocationT,
        VolumeSampleT,
        VolumeSamplingContextT,
        VolumeSampleProcessingContextT,
        VolumeT,
        VolumeProcessingT,
        VolumeProcessingContextT,
        VolumeSamplingSubdivisionProcessingWithSolidsBoundary<
                IndicesT,
                VolumeLocationT,
                VolumeSampleT,
                VolumeSamplingContextT,
                VolumeT,
                VolumeProcessingT
            >,
        VolumeSamplingSubdivisionProcessingContextWithSolidsBoundary<
                IndicesT,
                VolumeLocationT,
                VolumeSampleT,
                VolumeSamplingContextT,
                VolumeSampleProcessingContextT,
                VolumeProcessingContextT
            >
    > {
    init(context: VolumeSamplingSubdivisionProcessingContextWithSolidsBoundary<
                IndicesT,
                VolumeLocationT,
                VolumeSampleT,
                VolumeSamplingContextT,
                VolumeSampleProcessingContextT,
                VolumeProcessingContextT
            >): ProcessorInitialization {
        return {
            connections: {
                inputs: [["samples"]],
                outputs: [...groupPaths(BoundaryVolumeSamplingSubdivisionOctTreeGroupsTemplate)]
            }
        }
    }
    
    process(
        item: VolumeSamplingSubdivisionProcessingWithSolidsBoundary<
                IndicesT,
                VolumeLocationT,
                VolumeSampleT,
                VolumeSamplingContextT,
                VolumeT,
                VolumeProcessingT
            >,
        context: VolumeSamplingSubdivisionProcessingContextWithSolidsBoundary<
                IndicesT,
                VolumeLocationT,
                VolumeSampleT,
                VolumeSamplingContextT,
                VolumeSampleProcessingContextT,
                VolumeProcessingContextT
            >
    ): void {
        const subdivision = context[SubdivisionKey]
        const layer = subdivision.depth
        const parent_layer = layer - 1

        const invalid_layer = 255
        const invalid_localIndex = subdivision.invalid

        if (parent_layer === -1) {
            context[DualKey] = {
                cells: {
                    neighbors: new OctTreeReferences(subdivision.typedArray),
                    vertices: new OctTreeReferences(subdivision.typedArray),
                    lookup: {
                        corners: new OctTreeReferences(subdivision.typedArray),
                    }
                },
            }

            item[DualKey] = {
                cells: {
                    neighbors: {
                        layers: context[DualKey].cells.neighbors.layers.subdivide(0),
                        localIndices: context[DualKey].cells.neighbors.localIndices.subdivide(0),
                    },
                    vertices: {
                        layers: context[DualKey].cells.vertices.layers.subdivide(0),
                        localIndices: context[DualKey].cells.vertices.localIndices.subdivide(0),
                    },
                    lookup: {
                        corners: {
                            layers: context[DualKey].cells.lookup.corners.layers.subdivide(8).fill(invalid_layer),
                            localIndices: context[DualKey].cells.lookup.corners.localIndices.subdivide(8).fill(invalid_localIndex),
                        }
                    }
                },
            }
        }
        else if (parent_layer === 0) {
            const neighbors = context[DualKey].cells.neighbors.subdivide(6)
            const vertices = context[DualKey].cells.neighbors.subdivide(8)
            const lookup_corners = context[DualKey].cells.lookup.corners.subdivide(64)
            
            neighbors.layers.fill(invalid_layer)
            neighbors.localIndices.fill(invalid_localIndex)
            vertices.layers.fill(invalid_layer)
            vertices.localIndices.fill(invalid_localIndex)
            lookup_corners.layers.fill(invalid_layer)
            lookup_corners.localIndices.fill(invalid_localIndex)

            // for (let face = 0; face < 6; face++) {
            //     neighbors.layers[face] = invalid_layer
            //     neighbors.localIndices[face] = invalid_localIndex
            // }
            
            for (let corner = 0; corner < 8; corner++) {
                vertices.layers[corner] = 1
                vertices.localIndices[corner] = corner

                const corner_opposite = octTreeSubcellOpposite.triagonal(<OctTreeCell>corner)
                lookup_corners.layers[(8 * corner) | corner_opposite] = 1
                lookup_corners.localIndices[(8 * corner) | corner_opposite] = 0
            }

            item[DualKey] = {
                cells: {
                    neighbors,
                    vertices,
                    lookup: {
                        corners: lookup_corners
                    }
                }
            }
        }
        else {
            // determine number of new dual cells
            //

            // const parent_cells = {
            //     neighbors: {
            //         localIndices: context[DualKey].cells.neighbors.localIndices.layers[parent_layer],
            //         layers: context[DualKey].cells.neighbors.layers.layers[parent_layer]
            //     },
            //     vertices: {
            //         localIndices: context[DualKey].cells.vertices.localIndices.layers[parent_layer],
            //         layers: context[DualKey].cells.vertices.layers.layers[parent_layer]
            //     },
            // }

            const references_parents = subdivision.references.parents.layers[parent_layer]

            const number_subdivided_primary_cells = Math.floor(subdivision.layer_sizes.at(-1)! / 8)

            const dual_cells = context[DualKey].cells

            let number_new_dual_cells = 0

            // 1 interior cell + 6 adjacent cells + (4 diagonal cells/plane * 3 planes) = 19 cells
            const max_number_new_dual_cells = 19 * number_subdivided_primary_cells
            
            const new_dual_cells_vertices_layers = new Uint8Array(8 * max_number_new_dual_cells)
            const new_dual_cells_vertices_localIndices = new subdivision.typedArray(8 * max_number_new_dual_cells)
            
            const new_dual_cells_neighbors_layers = new Uint8Array(6 * max_number_new_dual_cells)
            const new_dual_cells_neighbors_localIndices = new subdivision.typedArray(6 * max_number_new_dual_cells)

            const new_dual_cells_lookup_corners = dual_cells.lookup.corners.subdivide(8 * number_subdivided_primary_cells)
            const { 
                layers: new_dual_cells_lookup_corners_layers,
                localIndices: new_dual_cells_lookup_corners_localIndices
            } = new_dual_cells_lookup_corners
            // the size is known for sure
            // const new_dual_cells_lookup_corners_layers = new Uint8Array(8 * number_subdivided_primary_cells)
            // const new_dual_cells_lookup_corners_localIndices = new subdivision.typedArray(8 * number_subdivided_primary_cells)

            new_dual_cells_vertices_layers.fill(invalid_layer)
            new_dual_cells_vertices_localIndices.fill(invalid_localIndex)

            new_dual_cells_neighbors_layers.fill(invalid_layer)
            new_dual_cells_neighbors_localIndices.fill(invalid_localIndex)

            new_dual_cells_lookup_corners_layers.fill(invalid_layer)
            new_dual_cells_lookup_corners_localIndices.fill(invalid_localIndex)

            const lookup_corners_parents_layers = dual_cells.lookup.corners.layers.layers[parent_layer]
            const lookup_corners_parents_localIndices = dual_cells.lookup.corners.localIndices.layers[parent_layer]

            // all new dual cells for adjacent neighbors must be on layer `layer`
            // const interior_dual_cells_neighbors_adjacent_layers = new subdivision.typedArray(6 * max_number_new_dual_cells)
            const interior_dual_cells_neighbors_adjacent_localIndices = new subdivision.typedArray(6 * max_number_new_dual_cells)
            
            // const primary_vertices_subdivided_pyramid_localIndices = new subdivision.typedArray(6 * number_subdivided_primary_cells)

            const primary_parent_neighbors_adjacent_localIndices = new subdivision.typedArray(6)
            const primary_parent_neighbors_adjacent_layers = new subdivision.typedArray(6)
            
            const primary_parent_address = new subdivision.typedArray(parent_layer)
            const primary_parent_neighbor_address = new subdivision.typedArray(parent_layer)

            // const plane_axes = new Uint8Array([
            //     // YZ
            //     1, 2,

            //     // ZX
            //     2, 0,

            //     // XY
            //     0, 1,
            // ])

            // const quadrant_directions = new Uint8Array([
            //     0, 0,
            //     1, 0,
            //     1, 1,
            //     0, 1,
            // ])

            // /**
            //  * This maps a triaongal 
            //  */
            // const triagonal_diagonal_neighbors = new Uint8Array([

            // ])

            /**
             * tmp_lookup_adjacent[(6 * (local_index on parent_layer)) + face] = local index of dual cell in new layer
             * Is this only needed for new pyramid dual cells?
             * face = (2 * axis) + direction = index in [x-, x+, y-, y+, z-, z+]
             */
            const tmp_lookup_adjacent = new subdivision.typedArray(6 * number_subdivided_primary_cells)

            /**
             * tmp_lookup_diagonal[(12 * (local_index on parent_layer)) + direction_diagonal] = local index of dual cell in new layer
             * 
             * diagonal_direction = (4 * plane) + quadrant
             */
            const tmp_lookup_diagonal = new subdivision.typedArray(12 * number_subdivided_primary_cells)
            tmp_lookup_diagonal.fill(invalid_localIndex)

            for (let primary_local_index_group = 0; primary_local_index_group < number_subdivided_primary_cells; primary_local_index_group++) {
                //TODO: OctTreeNeighbors class can optimize this

                const primary_layer = layer
                const primary_local_index_offset = 8 * primary_local_index_group

                const primary_parent_local_index = references_parents[primary_local_index_group]
                subdivision.address(parent_layer, primary_parent_local_index, primary_parent_address)

                // update triagonal neighbors
                // function find_neighbor(
                //     axis0: 0 | 1 | 2, direction0: -1 | 1,
                //     axis1: 0 | 1 | 2, direction1: -1 | 1,
                //     axis2: 0 | 1 | 2, direction2: -1 | 1) {
                    
                // }

                // const new_dual_cell_interior_layer = layer
                const new_dual_cell_interior_localIndex = number_new_dual_cells++

                for (let corner = 0; corner < 8; corner++) {
                    new_dual_cells_vertices_layers[(8 * new_dual_cell_interior_localIndex) + corner] = layer
                    new_dual_cells_vertices_localIndices[(8 * new_dual_cell_interior_localIndex) + corner] = primary_local_index_offset | corner
                    
                    const primary_child_localIndex = primary_local_index_offset | corner
                    const corner_opposite = 0b111 ^ corner
                    new_dual_cells_lookup_corners_layers[(8 * primary_child_localIndex) | corner_opposite] = layer // new_dual_cell_interior_layer
                    new_dual_cells_lookup_corners_localIndices[(8 * primary_child_localIndex) | corner_opposite] = new_dual_cell_interior_localIndex
                }

                primary_parent_neighbors_adjacent_layers.fill(invalid_layer)
                primary_parent_neighbors_adjacent_localIndices.fill(invalid_localIndex)

                // make/update adjacent dual cells
                for (let adjacent_direction = 0; adjacent_direction < 6; adjacent_direction++) {
                    const axis = <Axis>(adjacent_direction >> 1)
                    const direction = <Direction>(adjacent_direction & 1)

                    // focus = the primary cell that was subdivded

                    const primary_focus_boundary_subcells = octTreeCellsByDirection[axis][direction]

                    //TODO: optimize for only finding real layer and local index
